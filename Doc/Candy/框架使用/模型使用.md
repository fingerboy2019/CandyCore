## 模型基础

### 目录结构

模型文件存放于 `Application/模块名/Model` 目录下：

```
Application/
└── User/
    ├── Model/
    │   ├── UserModel.php      # 用户模型
    │   └── RoleModel.php      # 角色模型
    └── Controller/
    └── View/
```

### 命名规范

- **文件名**：使用大驼峰命名法（PascalCase）
  - `UserModel.php`
  - `ArticleModel.php`
  - `OrderItemModel.php`

- **类名**：与文件名保持一致
- **表名**：默认小写，使用下划线分隔
  - `UserModel` → `user` 表
  - `ArticleModel` → `article` 表

### 模型定义

模型需继承 `Candy\Core\Model` 基类，支持自定义表名、主键等配置：

```php
<?php
namespace App\User\Model;

use Candy\Core\Model;

class UserModel extends Model
{
    /**
     * 模型名称（用于缓存键等）
     * @var string
     */
    protected string $name = 'User';

    /**
     * 数据表名称（默认：user）
     * @var string
     */
    protected string $table = 'user';

    /**
     * 主键名称（默认：id）
     * @var string
     */
    protected string $primaryKey = 'id';

    /**
     * 自动时间戳（默认开启）
     * true: 自动维护 create_time/update_time
     * false: 关闭自动时间戳
     * @var bool
     */
    protected bool $autoTimestamp = true;

    /**
     * 时间戳字段名（默认：create_time, update_time）
     * @var array
     */
    protected array $timestampFields = ['create_time', 'update_time'];

    /**
     * 可见字段（null 表示所有字段）
     * @var array|null
     */
    protected ?array $visible = null;

    /**
     * 隐藏字段（默认隐藏 password, salt 等敏感字段）
     * @var array
     */
    protected array $hidden = ['password', 'salt'];

    /**
     * 追加字段（自动追加的虚拟字段）
     * @var array
     */
    protected array $appends = [];

    /**
     * 全局查询条件（每次查询自动应用）
     * @var array
     */
    protected array $where = [];

    /**
     * 模型事件（见下方详细说明）
     * @var array
     */
    protected array $events = [];
}
```

## 数据查询

### 基础查询

```php
// 查询单条记录
$user = UserModel::find(1);

// 查询多条记录
$users = UserModel::select();

// 指定字段查询
$names = UserModel::only(['id', 'username'])->select();

// 排除字段查询
$users = UserModel::except(['password', 'salt'])->select();

// 使用查询构建器
$user = Candy\DB\Connection::getInstance()
    ->table('User')
    ->where('status', 1)
    ->find();
```

### 条件查询

```php
// where 条件查询
UserModel::where('status', 1)->select();
UserModel::where('age', '>', 18)->select();
UserModel::where('name', 'like', '%张%')->select();

// 多个条件（AND）
UserModel::where('status', 1)
    ->where('age', '>', 18)
    ->select();

// 多个条件（OR）
UserModel::where('status', 1)
    ->orWhere('is_vip', 1)
    ->select();

// IN 查询
UserModel::whereIn('id', [1, 2, 3])->select();

// NOT IN 查询
UserModel::whereNotIn('id', [1, 2, 3])->select();

// EXISTS 子查询
UserModel::whereExists(function ($query) {
    $query->select('id')->from('orders')->where('user_id', UserModel::$primaryKey);
})->select();

// LIKE 查询
UserModel::whereLike('username', '%admin%')->select();

// BETWEEN 查询
UserModel::whereBetween('age', [18, 60])->select();

// NULL 查询
UserModel::whereNull('email')->select();
UserModel::whereNotNull('email')->select();
```

### 排序与分页

```php
// 排序
UserModel::order('create_time', 'desc')->select();
UserModel::order(['username' => 'asc', 'age' => 'desc'])->select();

// 限制数量
UserModel::limit(10)->select();
UserModel::limit(10, 0)->select(); // 偏移量

// 分页查询
$page = I('get.page', 1);
$pageSize = I('get.pageSize', 20);

$users = UserModel::page($page, $pageSize)->select();
$total = UserModel::count();

// 分配数据到视图
$this->assign('users', $users);
$this->assign('total', $total);
$this->assign('page', $page);
```

### 关联查询

```php
// 一对多关联（用户 - 文章）
class UserModel extends Model
{
    public function articles()
    {
        return $this->hasMany('App\Article\Model\ArticleModel', 'user_id', 'id');
    }
}

class ArticleModel extends Model
{
    public function user()
    {
        return $this->belongsTo('App\User\Model\UserModel', 'user_id', 'id');
    }
}

// 预加载关联数据
$user = UserModel::with('articles')->find(1);
$articles = $user->articles;  // 获取用户所有文章

// 指定预加载字段
$user = UserModel::with(['articles', 'articles.tags'])->find(1);

// 条件预加载
$user = UserModel::with(['articles' => function ($query) {
    $query->where('status', 1);
}])->find(1);
```

## 数据操作

### 新增数据

```php
// 方式 1：创建新实例
$user = new UserModel();
$user->username = 'test_user';
$user->email = 'test@example.com';
$user->age = 25;
$userId = $user->save();  // 返回新增的主键 ID

// 方式 2：静态方法
$user = UserModel::create([
    'username' => 'test_user',
    'email' => 'test@example.com',
    'age' => 25
]);

// 方式 3：使用数组
$data = [
    'username' => 'test_user',
    'email' => 'test@example.com',
    'age' => 25
];
$userId = UserModel::insert($data);

// 获取插入 ID
$lastId = UserModel::getLastInsertId();
```

### 更新数据

```php
// 方式 1：通过实例更新
$user = UserModel::find(1);
$user->age = 26;
$user->save();

// 方式 2：静态方法
UserModel::where('id', 1)->update([
    'age' => 26,
    'status' => 1
]);

// 方式 3：批量更新
UserModel::where('age', '<', 20)->update([
    'status' => 0
]);

// 方式 4：使用数组更新
$data = [
    'username' => 'new_name',
    'email' => 'new@example.com'
];
UserModel::update(1, $data);

// 方式 5：增量更新
UserModel::where('id', 1)->increment('score', 10);
UserModel::where('id', 1)->decrement('score', 5);
```

### 删除数据

```php
// 方式 1：通过实例删除
$user = UserModel::find(1);
$user->delete();

// 方式 2：静态方法
UserModel::destroy(1);
UserModel::destroy([1, 2, 3]);  // 批量删除

// 方式 3：条件删除
UserModel::where('status', 0)->delete();
UserModel::where('create_time', '<', '2020-01-01')->delete();

// 方式 4：使用 where 删除
UserModel::where('id', '>', 100)->delete();
```

## 高级功能

### 模型事件

支持在数据操作前后触发事件：

```php
class UserModel extends Model
{
    /**
     * 新增前事件
     */
    protected function beforeInsert()
    {
        if (empty($this->username)) {
            throw new \Exception('用户名不能为空');
        }

        // 自动生成密码
        if (empty($this->password)) {
            $this->password = md5(time());
        }
    }

    /**
     * 新增后事件
     */
    protected function afterInsert()
    {
        // 记录日志
        Log::info('用户新增', ['user_id' => $this->id]);

        // 发送欢迎邮件
        // Mail::send($user, 'welcome_email');
    }

    /**
     * 更新前事件
     */
    protected function beforeUpdate()
    {
        // 验证更新数据
        if (!empty($this->email) && !$this->isValidEmail($this->email)) {
            throw new \Exception('邮箱格式不正确');
        }
    }

    /**
     * 更新后事件
     */
    protected function afterUpdate()
    {
        // 清除用户缓存
        cache('user_' . $this->id, null);
    }

    /**
     * 删除前事件
     */
    protected function beforeDelete()
    {
        // 检查是否有子数据
        if (ArticleModel::where('user_id', $this->id)->count() > 0) {
            throw new \Exception('用户还有文章，无法删除');
        }
    }

    /**
     * 删除后事件
     */
    protected function afterDelete()
    {
        // 记录日志
        Log::info('用户删除', ['user_id' => $this->id]);
    }
}
```

### 字段可见性

```php
class UserModel extends Model
{
    /**
     * 可见字段（null 表示所有字段）
     */
    protected array $visible = ['id', 'username', 'email'];

    /**
     * 隐藏字段
     */
    protected array $hidden = ['password', 'salt'];

    /**
     * 追加字段（虚拟字段）
     */
    protected array $appends = ['full_name'];

    // 虚拟字段方法
    public function getFullNameAttr($value)
    {
        return $this->username . ' ' . $this->nickname;
    }
}

// 查询时只返回可见字段
$users = UserModel::select()->toArray();

// 获取隐藏字段需要指定
$users = UserModel::withHidden(['password'])->select();
```

### 软删除

```php
class UserModel extends Model
{
    /**
     * 软删除字段（默认：deleted_at）
     */
    protected string $deletedAt = 'deleted_at';

    /**
     * 启用软删除
     */
    protected bool $enableDeleteTimestamps = true;
}

// 软删除
UserModel::find(1)->delete();  // 不真正删除，只设置 deleted_at

// 查询时排除已删除数据
$users = UserModel::withTrashed()->select();

// 恢复已删除数据
UserModel::withTrashed()->find(1)->restore();
```

### 事务处理

```php
class OrderModel extends Model
{
    /**
     * 创建订单（事务处理）
     */
    public function createOrder($userId, $items)
    {
        try {
            // 开启事务
            Candy\DB\Connection::getInstance()->beginTransaction();

            // 1. 创建订单
            $order = new OrderModel();
            $order->user_id = $userId;
            $order->total = array_sum(array_column($items, 'price'));
            $order->save();

            // 2. 创建订单明细
            foreach ($items as $item) {
                $itemModel = new OrderItemModel();
                $itemModel->order_id = $order->id;
                $itemModel->product_id = $item['product_id'];
                $itemModel->price = $item['price'];
                $itemModel->quantity = $item['quantity'];
                $itemModel->save();
            }

            // 3. 扣减库存
            foreach ($items as $item) {
                ProductModel::where('id', $item['product_id'])
                    ->decrement('stock', $item['quantity']);
            }

            // 提交事务
            Candy\DB\Connection::getInstance()->commit();

            return $order;
        } catch (\Exception $e) {
            // 回滚事务
            Candy\DB\Connection::getInstance()->rollback();
            throw $e;
        }
    }
}
```

## 最佳实践

### 1. 使用查询构建器

```php
// ❌ 不好：硬编码 SQL
$user = DB::query("SELECT * FROM user WHERE id = {$id}");

// ✅ 好：使用查询构建器
$user = UserModel::find($id);
```

### 2. 参数化查询防止注入

```php
// ❌ 不好：字符串拼接
$user = UserModel::where("username = '{$username}'", '1')->find();

// ✅ 好：使用参数绑定
$user = UserModel::where('username', $username)->find();
```

### 3. 合理使用关联查询

```php
// ❌ 不好：N+1 查询问题
$users = UserModel::select()->toArray();
foreach ($users as $user) {
    $articles = ArticleModel::where('user_id', $user['id'])->select()->toArray();
}

// ✅ 好：预加载关联数据
$users = UserModel::with('articles')->select()->toArray();
```

### 4. 使用可见字段限制

```php
// ✅ 好：限制返回字段
$users = UserModel::only(['id', 'username', 'email'])->select();
```

### 5. 异常处理

```php
try {
    $user = UserModel::find($id);
    if (!$user) {
        throw new \Exception('用户不存在');
    }
} catch (\Exception $e) {
    Log::error($e->getMessage());
    $this->error('操作失败');
}
```
